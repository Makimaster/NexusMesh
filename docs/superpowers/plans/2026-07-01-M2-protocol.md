"""五阶段协议事件类型枚举与分阶段 payload 模型。"""
from __future__ import annotations

from enum import Enum

from pydantic import BaseModel, ConfigDict

from app.protocol.stages import ProtocolStage


class EventType(str, Enum):
    """标准 Agent 事件类型。继承 str 以支持 JSON 序列化。"""

    AGENT_SPAWN = "agent_spawn"
    AGENT_CALL = "agent_call"
    AGENT_FINISH = "agent_finish"
    AGENT_REFLECT = "agent_reflect"


# ---------------------------------------------------------------------------
# 分阶段 payload 模型（最小字段，按需扩展）
# ---------------------------------------------------------------------------


class _ProtocolPayload(BaseModel):
    """协议 payload 基类：拒绝未知字段，避免静默吞掉协议漂移。"""

    model_config = ConfigDict(extra="forbid")


class InitPayload(_ProtocolPayload):
    """INIT 阶段：建立 Agent 上下文。"""

    agent_id: str
    workflow_id: str


class ReceivePayload(_ProtocolPayload):
    """RECEIVE 阶段：接收任务输入。"""

    task_input: dict


class RoutePayload(_ProtocolPayload):
    """ROUTE 阶段：多跳路由决策。"""

    next_agent_id: str | None = None
    routing_reason: str | None = None


class ExecutePayload(_ProtocolPayload):
    """EXECUTE 阶段：LLM 调用结果，对齐设计文档 §8.3。"""

    agent_message: str
    prompt_tokens: int
    completion_tokens: int
    model: str
    model_cost_usd: float


class FinishPayload(_ProtocolPayload):
    """FINISH 阶段：写入执行记录。"""

    success: bool
    output: dict | None = None


# stage → payload 类的映射，供 serializer.py 与 ProtocolEvent validator 使用
_PAYLOAD_CLS_MAP: dict[ProtocolStage, type[BaseModel]] = {
    ProtocolStage.INIT: InitPayload,
    ProtocolStage.RECEIVE: ReceivePayload,
    ProtocolStage.ROUTE: RoutePayload,
    ProtocolStage.EXECUTE: ExecutePayload,
    ProtocolStage.FINISH: FinishPayload,
}
```

```python


# ---------------------------------------------------------------------------
# 统一信封
# ---------------------------------------------------------------------------


class ProtocolEvent(BaseModel):
    """五阶段协议统一信封。

    payload 必须是与 protocol_stage 匹配的具体 payload 模型实例；
    传错类型时 model_validator 抛 ValidationError，Fail-Fast。
    """

    model_config = ConfigDict(arbitrary_types_allowed=True, validate_assignment=True)

    protocol_stage: ProtocolStage
    event_type: EventType
    payload: Any  # 实际类型由 model_validator 强制校验
    created_at: datetime = Field(
        default_factory=lambda: datetime.now(timezone.utc)  # noqa: UP017
    )

    @field_validator("created_at", mode="after")
    @classmethod
    def _normalize_created_at(cls, value: datetime) -> datetime:
        """强制 created_at 为 UTC-aware。

        validate_assignment=True 使本校验对「构造后再赋值」同样生效，
        守住 to_payload 出站永远 UTC-aware 的不变量。
        """
        if value.tzinfo is None or value.utcoffset() is None:
            raise ValueError("created_at must be timezone-aware")
        return value.astimezone(UTC)

    @model_validator(mode="after")
    def _check_payload_matches_stage(self) -> ProtocolEvent:
        expected_cls = _PAYLOAD_CLS_MAP[self.protocol_stage]
        if not isinstance(self.payload, expected_cls):
            raise ValueError(
                f"stage {self.protocol_stage.value} 的 payload 必须是"
                f" {expected_cls.__name__}，实际传入 {type(self.payload).__name__}"
            )
        return self
```

```python
"""五阶段协议事件的扁平 payload 序列化器。

to_payload : ProtocolEvent → dict（写入 execution_events.payload JSONB）
from_payload: dict         → ProtocolEvent（从 JSONB 还原）
"""
from __future__ import annotations

from datetime import UTC, datetime
from typing import Any

from app.protocol.events import _PAYLOAD_CLS_MAP, EventType, ProtocolEvent
from app.protocol.stages import ProtocolStage

# to_payload 时需要排除、不放入 payload dict 的顶层信封字段
_ENVELOPE_KEYS = frozenset({"protocol_stage", "event_type", "created_at"})


def to_payload(event: ProtocolEvent) -> dict[str, Any]:
    """将协议事件序列化为协议传输用的扁平 dict。

    产物结构（以 EXECUTE 为例，对齐设计文档 §8.3）：
    {
        "agent_message": "...",
        "prompt_tokens": 512,
        "completion_tokens": 128,
        "model": "gpt-4o",
        "model_cost_usd": 0.0032,
        "protocol_stage": "EXECUTE",
        "event_type": "agent_finish",
        "created_at": "2026-07-01T12:00:00+00:00"
    }

    created_at 的 UTC-aware 不变量由 ProtocolEvent 校验器保证（含 validate_assignment），
    此处无需重复归一化。
    """
    return {
        **event.payload.model_dump(),
        "protocol_stage": event.protocol_stage.value,
        "event_type": event.event_type.value,
        "created_at": event.created_at.isoformat(),
    }


def from_payload(data: dict[str, Any]) -> ProtocolEvent:
    """从协议传输 payload 还原协议事件。

    根据 data["protocol_stage"] 选取对应的 payload 模型类；
    非法 stage 字符串抛 ValueError，naive 时间抛 ValueError，
    payload 字段缺失/未知字段抛 ValidationError。
    """
    stage = ProtocolStage(data["protocol_stage"])        # 非法值 → ValueError
    event_type = EventType(data["event_type"])           # 非法值 → ValueError
    created_at = datetime.fromisoformat(data["created_at"])
    if created_at.tzinfo is None or created_at.utcoffset() is None:
        raise ValueError("created_at must be timezone-aware")
    created_at = created_at.astimezone(UTC)

    payload_cls = _PAYLOAD_CLS_MAP[stage]
    payload_data = {
        key: value for key, value in data.items() if key not in _ENVELOPE_KEYS
    }
    payload = payload_cls(**payload_data)               # 缺/多字段 → ValidationError

    return ProtocolEvent(
        protocol_stage=stage,
        event_type=event_type,
        payload=payload,
        created_at=created_at,
    )
```

```markdown
# NexusMesh 五阶段协议规范

**版本**：1.0.0
**日期**：2026-07-01
**状态**：已实现（M2）
**实现位置**：`backend/app/protocol/`

---

## 1. 概述

所有 Agent 间通信必须经过 `protocol/` 模块封包。协议将一次 Agent 执行拆分为五个有序阶段，
每个阶段产出一或多个 `ProtocolEvent`，写入 `execution_events.payload` (JSONB)，
同时经 Redis Pub/Sub 实时推送前端。

---

## 2. 协议阶段

| 序号 | 阶段 | 枚举值 | 说明 |
|------|------|--------|------|
| 1 | 初始化 | `INIT` | 建立 Agent 上下文（agent_id、workflow_id） |
| 2 | 任务接收 | `RECEIVE` | 解析输入，准备 task_input |
| 3 | 路由决策 | `ROUTE` | 多跳路由，决定 next_agent_id |
| 4 | LLM 执行 | `EXECUTE` | 调用 LLM，产出 token 流与用量统计 |
| 5 | 结果返回 | `FINISH` | 写入执行记录，标记 success/output |

---

## 3. 标准事件类型

| 枚举值 | 字符串值 | 典型阶段 |
|--------|----------|---------|
| `AGENT_SPAWN` | `agent_spawn` | INIT |
| `AGENT_CALL` | `agent_call` | RECEIVE / ROUTE |
| `AGENT_FINISH` | `agent_finish` | EXECUTE / FINISH |
| `AGENT_REFLECT` | `agent_reflect` | 任意阶段 |

---

## 4. ProtocolEvent 信封结构

```python
ProtocolEvent(
    protocol_stage = ProtocolStage,   # 枚举，强校验
    event_type     = EventType,       # 枚举，强校验
    payload        = <stage payload>, # 必须与 stage 匹配，否则抛 ValidationError
    created_at     = datetime,        # UTC，默认 datetime.now(timezone.utc)
)
```

---

## 5. 各阶段 payload 字段

### INIT — InitPayload

| 字段 | 类型 | 必填 |
|------|------|------|
| `agent_id` | `str` | ✓ |
| `workflow_id` | `str` | ✓ |

### RECEIVE — ReceivePayload

| 字段 | 类型 | 必填 |
|------|------|------|
| `task_input` | `dict` | ✓ |

### ROUTE — RoutePayload

| 字段 | 类型 | 必填 |
|------|------|------|
| `next_agent_id` | `str \| None` | — |
| `routing_reason` | `str \| None` | — |

### EXECUTE — ExecutePayload（对齐设计文档 §8.3）

| 字段 | 类型 | 必填 | 说明 |
|------|------|------|------|
| `agent_message` | `str` | ✓ | LLM 完整输出 |
| `prompt_tokens` | `int` | ✓ | 输入 token 数 |
| `completion_tokens` | `int` | ✓ | 输出 token 数 |
| `model` | `str` | ✓ | 模型名（如 `gpt-4o`） |
| `model_cost_usd` | `float` | ✓ | 本次调用美元成本 |

### FINISH — FinishPayload

| 字段 | 类型 | 必填 |
|------|------|------|
| `success` | `bool` | ✓ |
| `output` | `dict \| None` | — |

## 6. JSONB 序列化格式

`serializer.to_payload()` 将信封铺平为 dict，存入 `execution_events.payload`：

```json
{
  "protocol_stage": "EXECUTE",
  "event_type": "agent_finish",
  "created_at": "2026-07-01T12:00:00+00:00",
  "agent_message": "分析完成",
  "prompt_tokens": 512,
  "completion_tokens": 128,
  "model": "gpt-4o",
  "model_cost_usd": 0.0032
}
```

`serializer.from_payload(data)` 根据 `data["protocol_stage"]` 选取 payload 模型，还原 `ProtocolEvent`。

- **M6 编排引擎是主要消费者**：字段扩展在 M6 开发时驱动，现阶段最小字段即可。
